/* ===================================== TRUSTNOVA BANK CUSTOMER REGISTRATION ===================================== */ document.addEventListener("DOMContentLoaded", () => { const registerForm = document.getElementById("registerForm"); if (!registerForm) { console.error("registerForm was not found."); return; } registerForm.addEventListener( "submit", handleRegistration ); }); async function handleRegistration(event) { event.preventDefault(); const email = document.getElementById("email")?.value.trim(); const password = document.getElementById("password")?.value; const confirmPassword = document.getElementById("confirmPassword")?.value; const message = document.getElementById("registerMessage"); if (!email || !password || !confirmPassword) { showRegisterMessage( message, "Please complete all required fields.", true ); return; } if (password !== confirmPassword) { showRegisterMessage( message, "Passwords do not match.", true ); return; } if (password.length < 8) { showRegisterMessage( message, "Password must be at least 8 characters.", true ); return; } setRegisterLoading(true); try { const { data, error } = await supabase.auth.signUp({ email, password }); if (error) { throw error; } /* * If email confirmation is enabled, * Supabase may return a user without * an active session. */ if (data?.user && !data?.session) { showRegisterMessage( message, "Registration successful. Please check your email to confirm your account.", false ); return; } if (data?.session) { showRegisterMessage( message, "Registration successful. Redirecting...", false ); window.location.href = "dashboard.html"; return; } showRegisterMessage( message, "Registration completed. Please check your email.", false ); } catch (error) { console.error( "Registration error:", error ); showRegisterMessage( message, getRegistrationErrorMessage(error), true ); } finally { setRegisterLoading(false); } } function setRegisterLoading( isLoading ) { const button = document.querySelector( '#registerForm button[type="submit"]' ); if (!button) { return; } button.disabled = isLoading; button.textContent = isLoading ? "Creating account..." : "Register"; } function showRegisterMessage( element, text, isError ) { if (!element) { return; } element.textContent = text; element.classList.toggle( "error", Boolean(isError) ); element.classList.toggle( "success", !isError ); } function getRegistrationErrorMessage( error ) { const message = String( error?.message || "" ).toLowerCase(); if ( message.includes("password") ) { return "Please choose a stronger password."; } if ( message.includes("invalid email") ) { return "Please enter a valid email address."; } if ( message.includes("rate limit") ) { return "Too many registration attempts. Please try again later."; } return "Unable to create your account. Please check your information and try again."; }
